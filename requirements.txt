# AppLiveCaption - Python 3.11 or 3.12, Windows 10/11 x64.
#
#     pip install -r requirements.txt
#
# Packages only this app uses are pinned to the tested versions. Packages that
# other software often shares (NumPy, huggingface_hub, CTranslate2, ...) take a
# range instead, so installing the app into an existing Python keeps the
# versions already there whenever they work. PyTorch is not needed.

# GUI
PySide6==6.12.0
PySide6-Fluent-Widgets[full]==1.11.3

# Speech recognition (faster-whisper / CTranslate2, Silero VAD via onnxruntime)
faster-whisper==1.2.1
# CTranslate2 4.6 bundles the cuDNN 9.1 loader; app/asr/cuda_libs.py pins the
# matching CUDA libraries.
ctranslate2>=4.6,<4.7
onnxruntime==1.31.0
tokenizers>=0.22,<1
# faster-whisper's file decoder doesn't work with PyAV 16+. The app never reads
# audio files with it, so the built exe leaves PyAV out (tools/pyi_rth_no_av.py).
av==15.1.0

# NVIDIA's CUDA libraries for the GPU are not listed: the app downloads them
# itself (Models page > NVIDIA GPU support, or "python -m app.cli cuda --download").
# Already installed nvidia-cublas-cu12 / nvidia-cudnn-cu12 packages are used too.

# Audio capture: WASAPI loopback, per-app process loopback (comtypes), app list (pycaw)
PyAudioWPatch==0.2.12.9
comtypes==1.4.17
pycaw==20260927
psutil>=7.2
soxr==1.1.0
numpy>=2.3,<3

# Model downloads
huggingface_hub>=1.11,<3
hf-xet>=1.4

# Simplified -> Traditional Chinese (OpenCC s2twp). The pure-Python port gives the
# same results as the OpenCC package and installs everywhere. Don't install both:
# they share the "opencc" folder and overwrite each other's files.
opencc-python-reimplemented>=0.1.7

# Packaging (only needed to build the .exe; 6.22 is the tested version).
pyinstaller>=6.22
pyinstaller-hooks-contrib>=2026.7
